from typing import List, Union, Optional, Any
from pydantic import BaseModel, Field, ConfigDict, model_validator

# UTILITY CLASSES (MANDATORY)

class SpatialCoordinatesPolygon(BaseModel):
    model_config = ConfigDict(extra='forbid')
    horizontal_x_vertices: List[float]
    vertical_y_vertices: List[float]

class ForensicDataEntity(BaseModel):
    model_config = ConfigDict(extra='forbid')
    extracted_string_or_numeric_value: Union[str, float]
    optical_extraction_confidence_score: float = Field(ge=0.0, le=1.0)
    physical_evidence_coordinates: SpatialCoordinatesPolygon

# HELPER FUNCTION FOR VALIDATOR

def get_val(entity: Optional[ForensicDataEntity]) -> float:
    """Safely extracts numeric value for calculations, returning 0.0 for None or non-numeric types."""
    if entity and isinstance(entity.extracted_string_or_numeric_value, (int, float)):
        return float(entity.extracted_string_or_numeric_value)
    return 0.0

# SCHEMA DEFINITION

class LineSummary(BaseModel):
    model_config = ConfigDict(extra='forbid')
    user_name: ForensicDataEntity
    phone_number: ForensicDataEntity
    plan_name: ForensicDataEntity
    monthly_charges: List[MonthlyChargeItem]
    total_monthly_charges: ForensicDataEntity
    total_usage_and_purchase_charges: ForensicDataEntity
    surcharges: LineSurcharges
    total_surcharges: ForensicDataEntity
    taxes_and_fees: LineTaxes
    total_taxes_fees: ForensicDataEntity
    total_current_charges_for_line: ForensicDataEntity
    call_details: Optional[List[CallDetail]] = None

    @model_validator(mode='after')
    def validate_line_charges(self) -> 'LineSummary':
        """Performs GAAP checksums for an individual line's charges."""
        # Check 1: Sum of itemized monthly charges equals total monthly charges
        sum_of_monthly_charges = sum(get_val(item.amount) for item in self.monthly_charges)
        if abs(sum_of_monthly_charges - get_val(self.total_monthly_charges)) > 0.01:
            raise ValueError("Line Summary: Sum of itemized monthly charges does not match total monthly charges.")

        # Check 2: Sum of all charge categories equals the total for the line
        total_line_charge_calc = (
            get_val(self.total_monthly_charges) +
            get_val(self.total_usage_and_purchase_charges) +
            get_val(self.total_surcharges) +
            get_val(self.total_taxes_fees)
        )
        if abs(total_line_charge_calc - get_val(self.total_current_charges_for_line)) > 0.01:
            raise ValueError("Line Summary: Sum of all charge categories does not match the total current charges for the line.")
            
        return self

class VerizonWirelessBill2014(BaseModel):
    model_config = ConfigDict(extra='forbid')
    account_number: ForensicDataEntity
    date_due: ForensicDataEntity
    invoice_number: ForensicDataEntity
    bill_date: ForensicDataEntity
    billing_period: ForensicDataEntity
    recipient_name: ForensicDataEntity
    recipient_address: ForensicDataEntity
    quick_summary: QuickBillSummary
    payments: PaymentsSummary
    account_charges_and_credits: AccountChargesSummary
    line_summaries: List[LineSummary]
    shared_usage: SharedUsageSummary
    total_charges_due: ForensicDataEntity
    remittance_slip: RemittanceSlip

    @model_validator(mode='after')
    def validate_bill_totals(self) -> 'VerizonWirelessBill2014':
        """Performs double-entry GAAP mathematical checksums across the entire bill."""
        # Check 1: Quick Summary Balance Forward calculation
        balance_forward_calc = get_val(self.quick_summary.previous_balance) + get_val(self.quick_summary.payment_thank_you)
        if abs(balance_forward_calc - get_val(self.quick_summary.balance_forward)) > 0.01:
            raise ValueError("Quick Summary: Previous Balance + Payment does not equal Balance Forward.")

        # Check 2: Quick Summary Total Current Charges calculation
        quick_summary_total_calc = (
            get_val(self.quick_summary.monthly_charges) +
            get_val(self.quick_summary.verizon_surcharges_and_other_charges_credits) +
            get_val(self.quick_summary.taxes_governmental_surcharges_fees)
        )
        if abs(quick_summary_total_calc - get_val(self.quick_summary.total_current_charges)) > 0.01:
            raise ValueError("Quick Summary: Sum of charges, surcharges, and taxes does not equal Total Current Charges.")

        # Check 3: Master cross-check of line items vs. grand total
        sum_of_line_charges = sum(get_val(line.total_current_charges_for_line) for line in self.line_summaries)
        account_level_charges = get_val(self.account_charges_and_credits.total_account_charges_and_credits)
        grand_total_calc = sum_of_line_charges + account_level_charges
        
        if abs(grand_total_calc - get_val(self.total_charges_due)) > 0.01:
            raise ValueError("Grand Total Mismatch: Sum of all line charges plus account charges does not equal the total amount due.")
            
        if abs(grand_total_calc - get_val(self.quick_summary.total_current_charges)) > 0.01:
            raise ValueError("Summary Mismatch: Sum of all line charges plus account charges does not equal the Quick Summary's total current charges.")

        return self
