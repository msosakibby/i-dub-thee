from pydantic import BaseModel, Field, ConfigDict, model_validator
from typing import List, Union, Optional
import math

class SpatialCoordinatesPolygon(BaseModel):
    """Defines the physical coordinates of an extracted data point on the document."""
    model_config = ConfigDict(extra='forbid')
    horizontal_x_vertices: List[float]
    vertical_y_vertices: List[float]

class ForensicDataEntity(BaseModel):
    """A wrapper for a single data point, including its value, confidence, and location."""
    model_config = ConfigDict(extra='forbid')
    extracted_string_or_numeric_value: Union[str, float]
    optical_extraction_confidence_score: float = Field(ge=0.0, le=1.0)
    physical_evidence_coordinates: SpatialCoordinatesPolygon

class TransactionItem(BaseModel):
    """A base model for a single transaction line item."""
    model_config = ConfigDict(extra='forbid')
    transaction_date: ForensicDataEntity
    posting_date: ForensicDataEntity
    description: ForensicDataEntity
    amount: ForensicDataEntity

class UpromiseStatement(BaseModel):
    """
    Schema for a Upromise by SallieMae MasterCard statement.
    This schema captures the earnings summary, payment activity, and a detailed list of purchases.
    """
    model_config = ConfigDict(extra='forbid')

    page_info: Optional[ForensicDataEntity] = None
    customer_name: ForensicDataEntity
    card_last_four: ForensicDataEntity
    
    # Earnings Summary
    earnings_this_period: ForensicDataEntity
    adjustments: ForensicDataEntity
    earnings_sent_to_upromise: ForensicDataEntity
    
    # Payment Activity
    total_payment_activity: ForensicDataEntity
    payments: List[TransactionItem]
    
    # Purchase Activity
    purchases: List[TransactionItem]

    @model_validator(mode='after')
    def validate_financial_checksums(self) -> 'UpromiseStatement':
        """
        Performs double-entry GAAP mathematical checksums on financial fields.
        1. Verifies that the earnings summary calculation is correct.
        2. Verifies that the total payment activity matches the sum of individual payments.
        """
        # Check 1: Earnings Summary Calculation
        earnings = self.earnings_this_period.extracted_string_or_numeric_value
        adjustments = self.adjustments.extracted_string_or_numeric_value
        sent_to_upromise = self.earnings_sent_to_upromise.extracted_string_or_numeric_value
        
        if not math.isclose(float(earnings) - float(adjustments), float(sent_to_upromise), rel_tol=1e-4):
            raise ValueError(
                f"Earnings summary mismatch: 'earnings_this_period' ({earnings}) - 'adjustments' ({adjustments}) "
                f"does not equal 'earnings_sent_to_upromise' ({sent_to_upromise})."
            )
            
        # Check 2: Total Payment Activity
        calculated_total_payments = sum(p.amount.extracted_string_or_numeric_value for p in self.payments)
        stated_total_payments = self.total_payment_activity.extracted_string_or_numeric_value
        
        if not math.isclose(calculated_total_payments, float(stated_total_payments), rel_tol=1e-4):
            raise ValueError(
                f"Total payment activity mismatch: Sum of payments ({calculated_total_payments}) "
                f"does not equal 'total_payment_activity' ({stated_total_payments})."
            )
            
        return self
