from typing import List, Union, Optional
from pydantic import BaseModel, ConfigDict, Field, model_validator

class SpatialCoordinatesPolygon(BaseModel):
    """Defines the spatial coordinates of an extracted entity on the document."""
    model_config = ConfigDict(extra='forbid')
    horizontal_x_vertices: List[float]
    vertical_y_vertices: List[float]

class ForensicDataEntity(BaseModel):
    """A wrapper for each extracted data point, including metadata."""
    model_config = ConfigDict(extra='forbid')
    extracted_string_or_numeric_value: Union[str, float]
    optical_extraction_confidence_score: float = Field(ge=0.0, le=1.0)
    physical_evidence_coordinates: SpatialCoordinatesPolygon

class Address(BaseModel):
    """A nested model for the holder's address."""
    model_config = ConfigDict(extra='forbid')
    street_and_number: ForensicDataEntity
    neighborhood: ForensicDataEntity
    postal_code: ForensicDataEntity
    municipality: ForensicDataEntity
    state_abbreviation: ForensicDataEntity

class MexicanVoterIdV1(BaseModel):
    """
    Schema for the Mexican 'Credencial para Votar' issued by the Instituto Nacional Electoral (INE).
    
    This model captures the personal, electoral, and identification details present on the front of the card.
    Document Class ID: 0196af14-6f62-43da-910e-5c2ec6702046
    """
    model_config = ConfigDict(extra='forbid')

    # Document Identification
    country: ForensicDataEntity
    issuing_authority: ForensicDataEntity
    document_title: ForensicDataEntity

    # Cardholder Information
    last_name: ForensicDataEntity
    second_last_name: ForensicDataEntity
    first_name: ForensicDataEntity
    date_of_birth: ForensicDataEntity
    sex: ForensicDataEntity
    address: Address

    # Unique Identifiers
    voter_key: ForensicDataEntity
    curp: ForensicDataEntity

    # Electoral Data
    electoral_section: ForensicDataEntity
    registration_year_and_month: ForensicDataEntity
    validity_period: ForensicDataEntity

    @model_validator(mode='after')
    def perform_gaap_checksums(self) -> 'MexicanVoterIdV1':
        """
        A mandatory validator for performing double-entry GAAP mathematical checksums.
        
        As this document contains no financial figures, this validator serves as a
        placeholder to fulfill the architectural requirement without performing calculations.
        """
        # No financial fields are present on this document for checksum validation.
        return self
