class DonationAcknowledgementLetter(BaseModel):
    """
    Schema for a donation acknowledgement letter, particularly for quid pro quo
    contributions where the donor receives goods or services.
    """

    event_date: Optional[ForensicDataEntity] = None

    signatory_title: Optional[ForensicDataEntity] = None

        """
        Validates financial consistency based on quid pro quo contribution rules.
        Total Payment = Value of Goods/Services Received + Deductible Contribution.
        This check only runs if all three relevant financial fields are present and numeric.
        """
        total = self.total_payment_amount
        cost = self.cost_of_event_per_ticket
        deductible = self.tax_deductible_amount

        if total and cost and deductible:

                    isinstance(deductible.extracted_string_or_numeric_value, (int, float))):
                # Cannot perform calculation if any value is not a number.

            total_val = total.extracted_string_or_numeric_value
            cost_val = cost.extracted_string_or_numeric_value
            deductible_val = deductible.extracted_string_or_numeric_value
            
            if not math.isclose(total_val, cost_val + deductible_val):
                raise ValueError(
                    f"Financial inconsistency: Total payment ({total_val}) must equal the sum of "
                    f"the cost of event ({cost_val}) and the tax deductible amount ({deductible_val})."
                )
                
        return self
