import math
from typing import List, Union, Optional
from pydantic import BaseModel, Field, ConfigDict, model_validator

class SpatialCoordinatesPolygon(BaseModel):
    model_config = ConfigDict(extra='forbid')
    horizontal_x_vertices: List[float]
    vertical_y_vertices: List[float]

class ForensicDataEntity(BaseModel):
    model_config = ConfigDict(extra='forbid')
    extracted_string_or_numeric_value: Union[str, float]
    optical_extraction_confidence_score: float = Field(ge=0.0, le=1.0)
    physical_evidence_coordinates: SpatialCoordinatesPolygon

def get_val(entity: ForensicDataEntity) -> float:
    """Helper to extract and clean numeric values from ForensicDataEntity."""
    val = entity.extracted_string_or_numeric_value
    if isinstance(val, str):
        return float(val.replace('$', '').replace(',', ''))
    return float(val)

class QuickBillSummary(BaseModel):

class VerizonWirelessBill(BaseModel):
    model_config = ConfigDict(extra='forbid')
    
    vendor_name: ForensicDataEntity
    account_number: ForensicDataEntity
    invoice_number: ForensicDataEntity
    bill_date: ForensicDataEntity
    due_date: ForensicDataEntity
    customer_name: ForensicDataEntity
    customer_address: ForensicDataEntity
    remittance_address: ForensicDataEntity
    quick_bill_summary: QuickBillSummary
    payments_summary: PaymentsSummary
    account_charges_and_credits: AccountChargesAndCredits
    account_plan: AccountPlan
    shared_usage_summary: SharedUsageSummary
    line_items: List[LineItem]
    total_amount_due: ForensicDataEntity
    remittance_amount_due: ForensicDataEntity

    @model_validator(mode='after')
    def validate_financials(self) -> 'VerizonWirelessBill':
        """Performs double-entry GAAP-style mathematical checksums."""
        
        # Check 1: Quick Bill Summary
        qbs = self.quick_bill_summary
        balance_forward_calc = get_val(qbs.previous_balance) + get_val(qbs.payment)
        if not math.isclose(balance_forward_calc, get_val(qbs.balance_forward)):
            raise ValueError("Quick Bill Summary: Previous Balance + Payment != Balance Forward")

        total_current_charges_calc = get_val(qbs.monthly_charges) + get_val(qbs.verizon_surcharges) + get_val(qbs.taxes_and_fees)
        if not math.isclose(total_current_charges_calc, get_val(qbs.total_current_charges)):
            raise ValueError("Quick Bill Summary: Sum of charges != Total Current Charges")

        # Check 2: Overall Total
        total_due_calc = get_val(qbs.balance_forward) + get_val(qbs.total_current_charges)
        if not math.isclose(total_due_calc, get_val(self.total_amount_due)):
            raise ValueError("Overall Total: Balance Forward + Total Current Charges != Total Amount Due")
        if not math.isclose(get_val(self.total_amount_due), get_val(self.remittance_amount_due)):
            raise ValueError("Overall Total: Total Amount Due != Remittance Amount Due")

        # Check 3: Line Item & Account Charges Summation
        total_line_items_charge = sum(get_val(line.total_current_charges_for_line) for line in self.line_items)
        account_charges = get_val(self.account_charges_and_credits.total)
        total_current_charges_from_breakdown = total_line_items_charge + account_charges
        if not math.isclose(total_current_charges_from_breakdown, get_val(qbs.total_current_charges)):
            raise ValueError("Sum of all line item totals + account charges != Total Current Charges")

        # Check 4: Individual Line Item Breakdowns
        for i, line in enumerate(self.line_items):
            line_total_calc = (
                get_val(line.total_monthly_charges) +
                get_val(line.total_usage_and_purchase_charges) +
                get_val(line.total_verizon_surcharges) +
                get_val(line.total_taxes_and_fees)
            )
            if not math.isclose(line_total_calc, get_val(line.total_current_charges_for_line)):
                raise ValueError(f"Line Item {i+1}: Sum of charge categories != line total")
            
            monthly_charges_calc = sum(get_val(c.amount) for c in line.previous_plan_adjustments) + \
                                   sum(get_val(c.amount) for c in line.new_plan_adjustments) + \
                                   sum(get_val(c.amount) for c in line.month_in_advance_charges)
            if not math.isclose(monthly_charges_calc, get_val(line.total_monthly_charges)):
                raise ValueError(f"Line Item {i+1}: Sum of monthly charges != total monthly charges")

            surcharges_sum = sum(get_val(item.amount) for item in line.verizon_surcharges)
            if not math.isclose(surcharges_sum, get_val(line.total_verizon_surcharges)):
                raise ValueError(f"Line Item {i+1}: Sum of surcharges != total surcharges")
                
            taxes_sum = sum(get_val(item.amount) for item in line.taxes_and_fees)
            if not math.isclose(taxes_sum, get_val(line.total_taxes_and_fees)):
                raise ValueError(f"Line Item {i+1}: Sum of taxes != total taxes")

        # Check 5: Account Charges
        account_charges_sum = sum(get_val(item.amount) for item in self.account_charges_and_credits.charges)
        if not math.isclose(account_charges_sum, get_val(self.account_charges_and_credits.subtotal)):
            raise ValueError("Account Charges: Sum of charges != subtotal")
        if not math.isclose(get_val(self.account_charges_and_credits.subtotal), get_val(self.account_charges_and_credits.total)):
            raise ValueError("Account Charges: Subtotal != total")

        # Check 6: Payments Summary
        payments_sum = sum(get_val(payment.amount) for payment in self.payments_summary.payments)
        if not math.isclose(payments_sum, get_val(self.payments_summary.total_payments)):
            raise ValueError("Payments Summary: Sum of payments != total payments")
        balance_forward_calc_2 = get_val(self.payments_summary.previous_balance) + get_val(self.payments_summary.total_payments)
        if not math.isclose(balance_forward_calc_2, get_val(self.payments_summary.balance_forward)):
            raise ValueError("Payments Summary: Previous Balance + Total Payments != Balance Forward")
            
        return self
