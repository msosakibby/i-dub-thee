class VerizonWirelessBillV1(BaseModel):
    model_config = ConfigDict(extra='forbid')
    account_number: ForensicDataEntity
    invoice_number: ForensicDataEntity
    date_due: ForensicDataEntity
    bill_date: ForensicDataEntity
    billing_period: ForensicDataEntity
    customer_name: ForensicDataEntity
    customer_address: ForensicDataEntity
    payment_address: ForensicDataEntity
    summary: BillSummary
    payments: PaymentDetails
    line_summaries: List[LineSummary]
    lines: List[LineDetail]
    remittance: RemittanceSlip

    @model_validator(mode='after')
    def perform_gaap_checksums(self) -> 'VerizonWirelessBillV1':
        """Performs double-entry GAAP mathematical checksums on financial data."""
        
        def v(entity: ForensicDataEntity) -> Decimal:
            return Decimal(str(entity.extracted_string_or_numeric_value))

        def vo(entity: Optional[ForensicDataEntity]) -> Decimal:
            return Decimal(str(entity.extracted_string_or_numeric_value)) if entity else Decimal(0)

        tolerance = Decimal('0.02')

        # --- Quick Bill Summary Validation ---
        summary = self.summary
        calc_balance_fwd = v(summary.previous_balance) + v(summary.payment_thank_you)
        assert abs(calc_balance_fwd - v(summary.balance_forward)) <= tolerance, "Summary Balance Forward mismatch"

        calc_current_charges = (v(summary.monthly_charges) + 
                                v(summary.usage_and_purchase_charges_voice) + 
                                v(summary.verizon_wireless_surcharges) + 
                                v(summary.taxes_governmental_surcharges_fees))
        assert abs(calc_current_charges - v(summary.total_current_charges)) <= tolerance, "Summary Total Current Charges mismatch"

        calc_total_due = v(summary.balance_forward) + v(summary.total_current_charges)
        assert abs(calc_total_due - v(summary.total_charges_due)) <= tolerance, "Summary Total Charges Due mismatch"
        assert abs(v(self.remittance.total_amount_due) - v(summary.total_charges_due)) <= tolerance, "Remittance amount mismatch"

        # --- Payment Details Validation ---
        payments = self.payments
        assert abs(v(payments.previous_balance) - v(summary.previous_balance)) <= tolerance, "Payment Details Previous Balance mismatch"
        calc_payment_balance_fwd = v(payments.previous_balance) + v(payments.total_payments)
        assert abs(calc_payment_balance_fwd - v(payments.balance_forward)) <= tolerance, "Payment Details Balance Forward mismatch"
        assert abs(v(payments.payment_received_amount) - v(payments.total_payments)) <= tolerance, "Payment Details Total Payments mismatch"

        # --- Line-Level and Cross-Total Validation ---
        total_monthly_from_lines = Decimal(0)
        total_usage_from_lines = Decimal(0)
        total_surcharges_from_lines = Decimal(0)
        total_taxes_from_lines = Decimal(0)

        for line in self.lines:
            # Line Monthly Charges
            calc_line_monthly = (vo(line.monthly_charges.plan_charge) + 
                                 vo(line.monthly_charges.addl_line_charge) + 
                                 vo(line.monthly_charges.equipment_coverage_charge))
            assert abs(calc_line_monthly - v(line.monthly_charges.total)) <= tolerance, f"Line {line.phone_number.extracted_string_or_numeric_value} monthly charges mismatch"
            
            # Line Surcharges
            calc_line_surcharges = (v(line.surcharges.fed_universal_service_charge) + 
                                    v(line.surcharges.regulatory_charge) + 
                                    v(line.surcharges.administrative_charge) + 
                                    v(line.surcharges.intrast_switched_toll_acc_sur))
            assert abs(calc_line_surcharges - v(line.surcharges.total)) <= tolerance, f"Line {line.phone_number.extracted_string_or_numeric_value} surcharges mismatch"

            # Line Taxes
            calc_line_taxes = (v(line.taxes.mi_state_911_charge) + 
                               v(line.taxes.osceola_cnty_911_charge) + 
                               v(line.taxes.mi_state_use_tax))
            assert abs(calc_line_taxes - v(line.taxes.total)) <= tolerance, f"Line {line.phone_number.extracted_string_or_numeric_value} taxes mismatch"

            # Total for Line
            calc_total_for_line = (v(line.monthly_charges.total) + 
                                   v(line.usage_charges.total) + 
                                   v(line.surcharges.total) + 
                                   v(line.taxes.total))
            assert abs(calc_total_for_line - v(line.total_current_charges_for_line)) <= tolerance, f"Line {line.phone_number.extracted_string_or_numeric_value} total current charges mismatch"

            # Aggregate totals for cross-check
            total_monthly_from_lines += v(line.monthly_charges.total)
            total_usage_from_lines += v(line.usage_charges.total)
            total_surcharges_from_lines += v(line.surcharges.total)
            total_taxes_from_lines += v(line.taxes.total)

        # --- Cross-Total Validation (Summary vs. Aggregated Lines) ---
        assert abs(total_monthly_from_lines - v(summary.monthly_charges)) <= tolerance, "Cross-total mismatch: Monthly Charges"
        assert abs(total_usage_from_lines - v(summary.usage_and_purchase_charges_voice)) <= tolerance, "Cross-total mismatch: Usage Charges"
        assert abs(total_surcharges_from_lines - v(summary.verizon_wireless_surcharges)) <= tolerance, "Cross-total mismatch: Surcharges"
        assert abs(total_taxes_from_lines - v(summary.taxes_governmental_surcharges_fees)) <= tolerance, "Cross-total mismatch: Taxes"

        return self
