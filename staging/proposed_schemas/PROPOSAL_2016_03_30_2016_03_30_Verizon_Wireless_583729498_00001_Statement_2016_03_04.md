from __future__ import annotations
import math
from typing import List, Union, Optional
from pydantic import BaseModel, ConfigDict, Field, model_validator, ValidationError

# Provided Base Classes
class SpatialCoordinatesPolygon(BaseModel):
    model_config = ConfigDict(extra='forbid')
    horizontal_x_vertices: List[float]
    vertical_y_vertices: List[float]

class ForensicDataEntity(BaseModel):
    model_config = ConfigDict(extra='forbid')
    extracted_string_or_numeric_value: Union[str, float]
    optical_extraction_confidence_score: float = Field(ge=0.0, le=1.0)
    physical_evidence_coordinates: SpatialCoordinatesPolygon

# Helper function for financial validation
def get_val(entity: Optional[ForensicDataEntity]) -> float:
    if entity is None:
        return 0.0
    val = entity.extracted_string_or_numeric_value
    if isinstance(val, str):
        # Clean string for float conversion (handles $, ,, -)
        return float(val.strip().replace('$', '').replace(',', ''))
    return float(val)

# Detailed Sub-Models

class VerizonWirelessBillV1(BaseModel):
    model_config = ConfigDict(extra='forbid')
    account_number: ForensicDataEntity
    invoice_number: ForensicDataEntity
    due_date: ForensicDataEntity
    billing_period: ForensicDataEntity
    quick_bill_summary: QuickBillSummary
    payment_details: PaymentDetails
    account_plan: AccountPlan
    account_level_charges: AccountLevelCharges
    line_details: List[LineDetail]
    shared_usage_breakdown: SharedUsageBreakdown
    total_charges_due: ForensicDataEntity
    remittance_slip: RemittanceSlip

    @model_validator(mode='after')
    def validate_financial_integrity(self) -> 'VerizonWirelessBillV1':
        errors = []

        # Summary Checks
        summary = self.quick_bill_summary
        if not math.isclose(get_val(summary.previous_balance) + get_val(summary.payment_received), get_val(summary.balance_forward)):
            errors.append("Summary: Previous Balance + Payment Received != Balance Forward")
        
        summary_charges_sum = get_val(summary.monthly_charges) + get_val(summary.verizon_surcharges) + get_val(summary.taxes_and_fees)
        if not math.isclose(summary_charges_sum, get_val(summary.total_current_charges)):
            errors.append("Summary: Sum of charges does not equal Total Current Charges")

        # Payment Details Checks
        payments = self.payment_details
        if not math.isclose(get_val(payments.previous_balance) + get_val(payments.payment_amount), get_val(payments.balance_forward)):
            errors.append("Payment Details: Previous Balance + Payment Amount != Balance Forward")

        # Account Level Charges Checks
        acc_charges = self.account_level_charges
        acc_charges_sum = sum(get_val(c.amount) for c in acc_charges.charges)
        if not math.isclose(acc_charges_sum, get_val(acc_charges.subtotal)):
            errors.append("Account Charges: Sum of itemized charges does not equal Subtotal")
        if not math.isclose(get_val(acc_charges.subtotal), get_val(acc_charges.total)):
            errors.append("Account Charges: Subtotal does not equal Total")

        # Line Item Checks
        total_line_charges = 0
        total_line_surcharges = 0
        total_line_taxes = 0
        total_line_monthly = 0
        for i, line in enumerate(self.line_details):
            line_sum = (get_val(line.monthly_charges.total) + get_val(line.usage_and_purchase_charges.total) +
                        get_val(line.verizon_surcharges.total) + get_val(line.taxes_and_fees.total))
            if not math.isclose(line_sum, get_val(line.total_current_charges_for_line)):
                errors.append(f"Line {i+1}: Sum of all charge categories does not equal Total Current Charges for Line")
            
            if not math.isclose(get_val(line.monthly_charges.smartphone_line_access) + get_val(line.monthly_charges.mobile_protection), get_val(line.monthly_charges.total)):
                errors.append(f"Line {i+1}: Sum of monthly charges does not equal monthly total")

            surcharge_sum = (get_val(line.verizon_surcharges.fed_universal_service) + get_val(line.verizon_surcharges.regulatory) +
                             get_val(line.verizon_surcharges.administrative) + get_val(line.verizon_surcharges.intrastate_switched_toll))
            if not math.isclose(surcharge_sum, get_val(line.verizon_surcharges.total)):
                errors.append(f"Line {i+1}: Sum of surcharges does not equal surcharges total")

            tax_sum = (get_val(line.taxes_and_fees.mi_state_911) + get_val(line.taxes_and_fees.osceola_county_911) +
                       get_val(line.taxes_and_fees.mi_state_use_tax))
            if not math.isclose(tax_sum, get_val(line.taxes_and_fees.total)):
                errors.append(f"Line {i+1}: Sum of taxes does not equal taxes total")
            
            total_line_charges += get_val(line.total_current_charges_for_line)
            total_line_surcharges += get_val(line.verizon_surcharges.total)
            total_line_taxes += get_val(line.taxes_and_fees.total)
            total_line_monthly += get_val(line.monthly_charges.total)

        # Cross-Component Checks
        if not math.isclose(total_line_charges + get_val(acc_charges.total), get_val(summary.total_current_charges)):
            errors.append("Cross-Component: Sum of all line charges + account charges != Summary Total Current Charges")
        
        if not math.isclose(total_line_surcharges, get_val(summary.verizon_surcharges)):
            errors.append("Cross-Component: Sum of line surcharges != Summary Surcharges")

        if not math.isclose(total_line_taxes, get_val(summary.taxes_and_fees)):
            errors.append("Cross-Component: Sum of line taxes != Summary Taxes & Fees")
        
        if not math.isclose(total_line_monthly + get_val(self.account_plan.monthly_charge), get_val(summary.monthly_charges)):
             errors.append("Cross-Component: Sum of line monthly charges + account plan charge != Summary Monthly Charges")

        # Final Total Checks
        if not math.isclose(get_val(summary.balance_forward) + get_val(summary.total_current_charges), get_val(self.total_charges_due)):
            errors.append("Final Total: Balance Forward + Total Current Charges != Total Charges Due")
        
        if not math.isclose(get_val(self.total_charges_due), get_val(self.remittance_slip.total_amount_due)):
            errors.append("Final Total: Total Charges Due does not match Remittance Slip Total Amount Due")

        if errors:
            raise ValueError("Financial integrity validation failed: " + "; ".join(errors))

        return self
