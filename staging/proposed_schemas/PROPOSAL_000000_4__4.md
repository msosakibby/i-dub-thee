class AmexCorporateCardStatement(BaseModel):
    model_config = ConfigDict(extra='forbid')
    
    prepared_for: ForensicDataEntity
    company_name: ForensicDataEntity
    account_number: ForensicDataEntity
    closing_date: ForensicDataEntity
    page_info: ForensicDataEntity
    previous_balance: ForensicDataEntity
    new_charges: ForensicDataEntity
    other_debits: ForensicDataEntity
    payments: ForensicDataEntity
    other_credits: ForensicDataEntity
    balance_due: ForensicDataEntity
    payment_due_date: ForensicDataEntity
    transactions: List[TransactionItem]
    payment_coupon: PaymentCoupon
    total_new_charges_other_debits: Optional[ForensicDataEntity] = None
    total_payments_other_credits: Optional[ForensicDataEntity] = None

    @model_validator(mode='after')
    def perform_gaap_checksums(self) -> 'AmexCorporateCardStatement':
        """
        Performs double-entry GAAP-based mathematical checksums.
        1. Validates the main summary balance equation.
        2. Validates that the sum of transaction items rolls up to the summary totals.
        3. Validates the optional final totals on the last page against the summary totals.
        """
        # Check 1: Summary block calculation
        summary_total = (
            self.previous_balance.extracted_string_or_numeric_value
            + self.new_charges.extracted_string_or_numeric_value
            + self.other_debits.extracted_string_or_numeric_value
            - self.payments.extracted_string_or_numeric_value
            - self.other_credits.extracted_string_or_numeric_value
        )
        if abs(summary_total - self.balance_due.extracted_string_or_numeric_value) > 0.01:
            raise ValueError("Balance due does not match summary calculation: Previous + New Charges + Debits - Payments - Credits != Balance Due")

        # Check 2: Transaction list summation vs. summary totals
        total_charges_from_transactions = sum(
            item.amount.extracted_string_or_numeric_value
            for item in self.transactions
            if item.amount.extracted_string_or_numeric_value > 0
        )
        total_credits_from_transactions = sum(
            item.amount.extracted_string_or_numeric_value
            for item in self.transactions
            if item.amount.extracted_string_or_numeric_value < 0
        )

        summary_charges = self.new_charges.extracted_string_or_numeric_value + self.other_debits.extracted_string_or_numeric_value
        summary_credits = -(self.payments.extracted_string_or_numeric_value + self.other_credits.extracted_string_or_numeric_value)

        if abs(total_charges_from_transactions - summary_charges) > 0.01:
            raise ValueError("Sum of positive transactions does not match 'New Charges' + 'Other Debits' in summary.")

        if abs(total_credits_from_transactions - summary_credits) > 0.01:
            raise ValueError("Sum of negative transactions does not match 'Payments' + 'Other Credits' in summary.")

        # Check 3: Optional final totals on last page
        if self.total_new_charges_other_debits:
            if abs(self.total_new_charges_other_debits.extracted_string_or_numeric_value - summary_charges) > 0.01:
                raise ValueError("Final total 'New Charges/Other Debits' does not match summary charges.")
        
        if self.total_payments_other_credits:
            if abs(self.total_payments_other_credits.extracted_string_or_numeric_value - summary_credits) > 0.01:
                raise ValueError("Final total 'Payments/Other Credits' does not match summary credits.")

        return self
