class VerizonWirelessBill(BaseModel):
    model_config = ConfigDict(extra='forbid')
    account_number: ForensicDataEntity
    invoice_number: ForensicDataEntity
    date_due: ForensicDataEntity
    bill_date: ForensicDataEntity
    customer_name: ForensicDataEntity
    customer_address: ForensicDataEntity
    billing_period: ForensicDataEntity
    quick_bill_summary: QuickBillSummary
    payments: Payments
    account_charges_and_credits: AccountChargesAndCredits
    plan_details: PlanDetails
    line_breakdowns: List[LineBreakdown]
    shared_usage_breakdown: SharedUsageBreakdown
    total_amount_due: ForensicDataEntity
    remittance_slip: RemittanceSlip

    @model_validator(mode='after')
    def perform_gaap_checksums(self) -> 'VerizonWirelessBill':
        """
        Performs double-entry GAAP mathematical checksums on financial data.
        A tolerance of 0.01 is used for all comparisons.
        """
        tolerance = Decimal('0.01')
        
        # Check 1: Quick Bill Summary internal consistency
        summary = self.quick_bill_summary
        calculated_current_charges = (
            summary.monthly_charges.to_decimal() +
            summary.surcharges_and_other.to_decimal() +
            summary.taxes_and_fees.to_decimal()
        )
        if abs(calculated_current_charges - summary.total_current_charges.to_decimal()) > tolerance:
            raise ValueError(f"Quick Bill Summary Check Failed: Calculated Total Current Charges ({calculated_current_charges:.2f}) does not match stated value ({summary.total_current_charges.to_decimal():.2f})")

        # Check 2: Total Amount Due calculation
        calculated_total_due = summary.balance_forward.to_decimal() + summary.total_current_charges.to_decimal()
        if abs(calculated_total_due - self.total_amount_due.to_decimal()) > tolerance:
            raise ValueError(f"Total Amount Due Check Failed: Calculated Total Due ({calculated_total_due:.2f}) does not match stated value ({self.total_amount_due.to_decimal():.2f})")

        # Check 3: Sum of all charges (Account + Lines) vs. Total Current Charges
        total_line_charges = sum(line.total_current_charges_for_line.to_decimal() for line in self.line_breakdowns)
        total_account_charges = self.account_charges_and_credits.total_account_charges_and_credits.to_decimal()
        
        sum_of_all_charges = total_account_charges + total_line_charges
        if abs(sum_of_all_charges - summary.total_current_charges.to_decimal()) > tolerance:
            raise ValueError(f"Overall Charge Summation Failed: Sum of account and line charges ({sum_of_all_charges:.2f}) does not match Total Current Charges ({summary.total_current_charges.to_decimal():.2f})")

        # Check 4: Individual Line Item breakdowns
        for line in self.line_breakdowns:
            calculated_line_total = (
                line.monthly_charges.total.to_decimal() +
                line.usage_and_purchase_charges.total.to_decimal() +
                line.surcharges.total.to_decimal() +
                line.taxes_and_fees.total.to_decimal()
            )
            if abs(calculated_line_total - line.total_current_charges_for_line.to_decimal()) > tolerance:
                raise ValueError(f"Line Breakdown Check Failed for {line.phone_number.extracted_string_or_numeric_value}: Calculated total ({calculated_line_total:.2f}) does not match stated total ({line.total_current_charges_for_line.to_decimal():.2f})")

        return self
