    model_config = ConfigDict(extra='forbid')
    customer_name: ForensicDataEntity
    customer_address: ForensicDataEntity
    account_number: ForensicDataEntity
    invoice_number: ForensicDataEntity
    bill_date: ForensicDataEntity

    line_items: List[LineItemDetail]
    total_amount_due: ForensicDataEntity

    @model_validator(mode='after')
    def validate_financials(self) -> 'VerizonWirelessBillV1':
        """Performs double-entry GAAP-style mathematical checksums on the bill."""
        
        def get_value(entity: ForensicDataEntity) -> float:
            val = entity.extracted_string_or_numeric_value
            if isinstance(val, str):
                val = val.replace('$', '').replace(',', '').strip()
                if val.startswith('(') and val.endswith(')'):
                    val = '-' + val[1:-1]
                return float(val)
            return float(val)

        # Check 1: Bill Summary calculations
        summary_prev_bal = get_value(self.summary.previous_balance)
        summary_payments = get_value(self.summary.payments_thank_you)
        summary_bal_fwd = get_value(self.summary.balance_forward)
        
        assert abs(summary_prev_bal + summary_payments - summary_bal_fwd) < 0.01, "Summary Balance Forward calculation is incorrect."

        summary_monthly = get_value(self.summary.monthly_charges)
        summary_surcharges = get_value(self.summary.surcharges_and_other_charges_credits)
        summary_taxes = get_value(self.summary.taxes_governmental_surcharges_fees)
        summary_total_current = get_value(self.summary.total_current_charges)
        
        calculated_summary_total = summary_bal_fwd + summary_monthly + summary_surcharges + summary_taxes
        assert abs(calculated_summary_total - summary_total_current) < 0.01, "Summary Total Current Charges calculation is incorrect."

        # Check 2: Account Level Charges calculations
        account_charges_subtotal = sum(get_value(charge.amount) for charge in self.account_level_charges.charges)
        assert abs(account_charges_subtotal - get_value(self.account_level_charges.subtotal)) < 0.01, "Account Level Charges subtotal is incorrect."
        assert abs(account_charges_subtotal - get_value(self.account_level_charges.total_account_charges_and_credits)) < 0.01, "Account Level Charges total is incorrect."

        # Check 3: Line Item calculations
        total_monthly_charges_from_lines = 0
        total_surcharges_from_lines = 0
        total_taxes_from_lines = 0
        total_current_charges_from_lines = 0

        for line in self.line_items:
            line_monthly_total = sum(get_value(charge.amount) for charge in line.monthly_charges.charges)
            assert abs(line_monthly_total - get_value(line.monthly_charges.total_monthly_charges)) < 0.01, f"Line item {get_value(line.phone_number)} monthly charges total is incorrect."
            
            line_surcharges_total = sum(get_value(item.amount) for item in line.surcharges.items)
            assert abs(line_surcharges_total - get_value(line.surcharges.total_surcharges)) < 0.01, f"Line item {get_value(line.phone_number)} surcharges total is incorrect."

            line_taxes_total = sum(get_value(item.amount) for item in line.taxes_and_fees.items)
            assert abs(line_taxes_total - get_value(line.taxes_and_fees.total_taxes)) < 0.01, f"Line item {get_value(line.phone_number)} taxes total is incorrect."

            line_usage_total = get_value(line.usage_and_purchase_charges.total_usage_and_purchase_charges)
            calculated_line_total = line_monthly_total + line_usage_total + line_surcharges_total + line_taxes_total
            assert abs(calculated_line_total - get_value(line.total_current_charges_for_line)) < 0.01, f"Line item {get_value(line.phone_number)} total current charges calculation is incorrect."

            total_monthly_charges_from_lines += line_monthly_total
            total_surcharges_from_lines += line_surcharges_total
            total_taxes_from_lines += line_taxes_total
            total_current_charges_from_lines += get_value(line.total_current_charges_for_line)

        # Check 4: Cross-component validation (Summary vs. detailed sections)
        calculated_total_monthly = get_value(self.account_level_charges.subtotal) + total_monthly_charges_from_lines
        assert abs(calculated_total_monthly - summary_monthly) < 0.01, "Summary monthly charges do not match sum of detailed monthly charges."

        assert abs(total_surcharges_from_lines - summary_surcharges) < 0.01, "Summary surcharges do not match sum of line item surcharges."
        assert abs(total_taxes_from_lines - summary_taxes) < 0.01, "Summary taxes do not match sum of line item taxes."

        calculated_total_from_components = get_value(self.account_level_charges.total_account_charges_and_credits) + total_current_charges_from_lines
        assert abs(calculated_total_from_components - summary_total_current) < 0.01, "Summary total current charges do not match sum of account and line item totals."

        # Check 5: Final Amount Due
        total_due = get_value(self.total_amount_due)
        calculated_total_due = summary_bal_fwd + summary_total_current
        assert abs(calculated_total_due - total_due) < 0.01, "Total Amount Due does not match Balance Forward + Total Current Charges."

        return self
